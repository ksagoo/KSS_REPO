#!/usr/bin/env python3

import argparse
from akamai_cert_manager_proxy import setup_session

parser = argparse.ArgumentParser()
parser.add_argument("--section", default="GLOBAL")
parser.add_argument("--enrollment-id", type=int, required=True)
args = parser.parse_args()

session, base_url = setup_session(
    edgerc_file="~/.edgerc",
    section=args.section
)

endpoint = f"cps/v2/enrollments/{args.enrollment_id}"

response = session.get(
    f"{base_url.rstrip('/')}/{endpoint}",
    headers={
        "Accept": "application/vnd.akamai.cps.enrollment.v2+json"
    },
    timeout=(10, 30)
)

print(f"Enrollment HTTP Status: {response.status_code}")
response.raise_for_status()

enrollment_json = response.json()

pending = enrollment_json.get("pendingChanges", []) or []

print(f"Pending Changes Found: {len(pending)}")

change_id = None

for pc in pending:
    location = pc.get("location", "")
    print(f"Pending Change Location: {location}")

    if "/changes/" in location:
        candidate = location.split("/changes/")[-1].strip()

        if candidate.isdigit():
            change_id = int(candidate)
            break

print(f"Discovered Change ID: {change_id}")

if change_id == 9230881:
    print("SUCCESS: Existing pending change discovered correctly.")
else:
    print("WARNING: Existing change 9230881 was not discovered.")

print("READ ONLY: No deployment scheduling performed.")
