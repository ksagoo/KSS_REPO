import sys

from akamai_cert_manager_proxy import (
    AkamaiCertManager,
    setup_session,
)

if len(sys.argv) != 3:
    print("Usage: python3 test_fqdn_lookup.py SECTION FQDN")
    sys.exit(1)

section = sys.argv[1]
fqdn = sys.argv[2]

session, base_url = setup_session(
    "~/.edgerc",
    section,
    False
)

if not session:
    print("ERROR: Failed to establish Akamai session")
    sys.exit(1)

manager = AkamaiCertManager(
    session,
    base_url,
    debug=False
)

enrollment_id = manager.get_enrollment_id_by_fqdn(fqdn)

if enrollment_id is None:
    print(f"ERROR: No unique enrollment found for {fqdn}")
    sys.exit(1)

print(f"SUCCESS: {fqdn.strip()} -> Enrollment ID {enrollment_id}")


python3 test_fqdn_lookup.py DEV "  YOUR.FQDN.COM  "
