
#!/usr/bin/env python3
"""Read-only Akamai CPS status diagnostic.

Uses the existing authentication helper from akamai_cert_manager_proxy.py.
No approval, renewal, scheduling or deployment requests are made.
"""

import argparse
import json
import sys
from datetime import datetime, timezone
from urllib.parse import quote

from akamai_cert_manager_proxy import setup_session

ACCEPT_ENROLLMENT = "application/vnd.akamai.cps.enrollment.v2+json"
ACCEPT_CHANGES = "application/vnd.akamai.cps.changes.v2+json"
ACCEPT_CHANGE = "application/vnd.akamai.cps.change.v2+json"
ACCEPT_DEPLOYMENTS = "application/vnd.akamai.cps.deployments.v2+json"


def get_json(session, base_url, path, accept):
    url = f"{base_url.rstrip('/')}/{path.lstrip('/')}"

    try:
        response = session.get(
            url,
            headers={"Accept": accept},
            timeout=(10, 60)
        )

        result = {
            "http_status": response.status_code,
            "ok": response.ok
        }

        try:
            result["body"] = response.json()
        except ValueError:
            result["body"] = response.text[:3000]

        return result

    except Exception as exc:
        return {
            "ok": False,
            "http_status": None,
            "error": str(exc)
        }


def find_changes(payload):
    if isinstance(payload, list):
        return payload

    if isinstance(payload, dict):
        for key in ("changes", "items"):
            if isinstance(payload.get(key), list):
                return payload[key]

    return []


def compact_change(change):
    if not isinstance(change, dict):
        return {"raw": change}

    fields = (
        "changeId",
        "id",
        "status",
        "state",
        "phase",
        "type",
        "createdDate",
        "lastUpdated",
        "deploymentSchedule",
        "deploymentDate",
        "scheduledDate",
        "scheduledTime",
        "production",
        "staging",
        "statusInfo"
    )

    result = {
        k: change[k]
        for k in fields
        if k in change
    }

    return result or change


def main():
    parser = argparse.ArgumentParser(
        description="Read-only Akamai CPS status checker"
    )

    parser.add_argument(
        "--section",
        default="DEV",
        help="EdgeRC section, e.g. GLOBAL"
    )

    parser.add_argument(
        "--edgerc-file",
        default="~/.edgerc"
    )

    parser.add_argument(
        "--enrollment-id",
        required=True
    )

    parser.add_argument(
        "--raw",
        action="store_true",
        help="Include full API responses"
    )

    args = parser.parse_args()

    enrollment_id = quote(
        str(args.enrollment_id),
        safe=""
    )

    session, base_url = setup_session(
        args.edgerc_file,
        args.section
    )

    prefix = f"cps/v2/enrollments/{enrollment_id}"

    enrollment = get_json(
        session,
        base_url,
        prefix,
        ACCEPT_ENROLLMENT
    )

    deployments = get_json(
        session,
        base_url,
        prefix + "/deployments",
        ACCEPT_DEPLOYMENTS
    )

    changes = get_json(
        session,
        base_url,
        prefix + "/changes",
        ACCEPT_CHANGES
    )

    result = {
        "checked_at_utc": datetime.now(
            timezone.utc
        ).isoformat(),

        "enrollment_id": args.enrollment_id,

        "http_status": {
            "enrollment": enrollment.get("http_status"),
            "deployments": deployments.get("http_status"),
            "changes": changes.get("http_status")
        },

        "changes_lookup": (
            "successful"
            if changes["ok"]
            else "failed"
        ),

        "warning": None,
        "changes": []
    }

    if not changes["ok"]:
        result["warning"] = (
            "Changes API lookup failed. "
            "This does NOT establish that "
            "there is no pending renewal. "
            "See changes_response."
        )

        result["changes_response"] = changes.get(
            "body",
            changes.get("error")
        )

    else:
        discovered = find_changes(
            changes.get("body")
        )

        for change in discovered:
            item = compact_change(change)

            change_id = (
                change.get("changeId") or change.get("id")
            ) if isinstance(change, dict) else None

            if change_id is not None:
                detail = get_json(
                    session,
                    base_url,
                    prefix + "/changes/" + quote(
                        str(change_id),
                        safe=""
                    ),
                    ACCEPT_CHANGE
                )

                item["detail_http_status"] = detail.get(
                    "http_status"
                )

                if detail["ok"]:
                    item["detail"] = (
                        detail["body"]
                        if args.raw
                        else compact_change(detail["body"])
                    )
                else:
                    item["detail_error"] = detail.get(
                        "body",
                        detail.get("error")
                    )

            result["changes"].append(item)

        if not discovered:
            result["warning"] = (
                "Changes API returned no entries; "
                "compare with CPS GUI before concluding "
                "there is no active renewal."
            )

    if args.raw:
        result["enrollment_response"] = enrollment
        result["deployments_response"] = deployments
        result["changes_response"] = changes

    else:
        for name, response in (
            ("enrollment", enrollment),
            ("deployments", deployments)
        ):
            if not response["ok"]:
                result[name + "_error"] = response.get(
                    "body",
                    response.get("error")
                )

        # Actual deployment response, not desired status.
        if deployments["ok"]:
            result["deployments"] = deployments.get("body")

        if enrollment["ok"] and isinstance(
            enrollment.get("body"), dict
        ):
            body = enrollment["body"]

            result["enrollment"] = {
                k: body[k]
                for k in (
                    "id",
                    "enrollmentId",
                    "status",
                    "validationType",
                    "certificateType",
                    "csr",
                    "networkConfiguration"
                )
                if k in body
            }

    print(
        "AKAMAI CPS CERTIFICATE STATUS (READ ONLY)"
    )

    print(
        json.dumps(
            result,
            indent=2,
            default=str
        )
    )

    return (
        0
        if all((
            enrollment["ok"],
            deployments["ok"],
            changes["ok"]
        ))
        else 1
    )


if __name__ == "__main__":
    sys.exit(main())
