    sections = list(dict.fromkeys([
        args.section, "KSS-GLOBAL", "APAC", "AMER", "EMEA"
    ]))

    for section in sections:
        print(f"\n[INFO] Trying EdgeGrid section: {section}")

        try:
            session, base_url = setup_session(
                args.edgerc_file,
                section,
                False,
            )

            if session is None or not base_url:
                print(f"[WARN] Session unavailable for {section}")
                continue

            result = lookup(
                session,
                base_url,
                args.fqdn,
                args.network,
            )

            if result == 0:
                print(f"[SUCCESS] Match found in {section}")
                return 0

            if result == 2 and args.network == "PRODUCTION":
                print(f"[INFO] Trying STAGING in {section}")

                result = lookup(
                    session,
                    base_url,
                    args.fqdn,
                    "STAGING",
                )

                if result == 0:
                    print(f"[SUCCESS] STAGING match in {section}")
                    return 0

            if result not in (0, 2):
                print(f"[ERROR] Lookup returned {result} in {section}")
                return result

        except Exception as exc:
            print(f"[WARN] Section {section} failed: {exc}")
            continue

    print("[ERROR] No matching FQDN found across EdgeGrid sections")
    return 2
