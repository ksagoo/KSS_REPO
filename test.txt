
#!/usr/bin/env python3
"""Read-only Akamai PAPI hostname -> property group lookup."""

import argparse
import sys
from urllib.parse import urljoin

from akamai_cert_manager_proxy import setup_session


def api_get(session, base_url, path, params=None):
    url = urljoin(base_url.rstrip("/") + "/", path.lstrip("/"))

    response = session.get(
        url,
        headers={
            "Accept": "application/json",
            "PAPI-Use-Prefixes": "true",
        },
        params=params,
        timeout=45,
    )

    response.raise_for_status()
    return response.json()


def walk_groups(items):
    for item in items:
        yield item
        yield from walk_groups(item.get("children") or [])


def lookup(session, base_url, fqdn, network):
    target = fqdn.strip().rstrip(".").casefold()

    if not target or any(c.isspace() for c in target):
        raise ValueError("Provide a valid non-empty FQDN")

    params = {
        "hostname": target,
        "network": network,
        "limit": 100,
    }

    path = "papi/v1/hostnames"
    seen_pages = set()
    matches = []

    # Search hostnames and follow pagination.
    while path:
        if path in seen_pages:
            raise RuntimeError("Repeated pagination link from PAPI")

        seen_pages.add(path)

        data = api_get(
            session,
            base_url,
            path,
            params=params,
        )

        hostnames = data.get("hostnames")

        if (
            not isinstance(hostnames, dict)
            or not isinstance(hostnames.get("items"), list)
        ):
            raise RuntimeError(
                "Unexpected PAPI hostnames response structure"
            )

        for item in hostnames["items"]:
            hostname = (
                item.get("cnameFrom")
                or item.get("hostname")
                or ""
            ).rstrip(".").casefold()

            if hostname == target:
                matches.append(item)

        path = hostnames.get("nextLink")
        params = None

    if not matches:
        print(
            f"No {network} hostname match found for "
            f"{target} among accessible properties."
        )
        return 2

    # Retrieve PAPI groups instead of IAM groups.
    group_data = api_get(
        session,
        base_url,
        "papi/v1/groups",
    )

    groups = group_data.get("groups")

    if (
        not isinstance(groups, dict)
        or not isinstance(groups.get("items"), list)
    ):
        raise RuntimeError(
            "Unexpected PAPI groups response structure"
        )

    by_id = {
        str(group.get("groupId")): group
        for group in walk_groups(groups["items"])
    }

    unresolved = False
    seen = set()

    for item in matches:
        key = (
            str(item.get("propertyId")),
            str(item.get("groupId")),
        )

        if key in seen:
            continue

        seen.add(key)

        group_id = str(item.get("groupId") or "")
        group = by_id.get(group_id)

        group_name = (
            group.get("groupName")
            if group
            else None
        )

        if not group_id or not group_name:
            unresolved = True

        print(f"\nFQDN         : {target}")
        print(
            f"Property     : "
            f"{item.get('propertyName') or 'Not provided'}"
        )
        print(
            f"Property ID  : "
            f"{item.get('propertyId') or 'Not provided'}"
        )
        print(
            f"Access Group : "
            f"{group_name or 'UNRESOLVED'}"
        )
        print(
            f"Group ID     : "
            f"{group_id or 'Not provided'}"
        )
        print(
            f"Contract ID  : "
            f"{item.get('contractId') or 'Not provided'}"
        )
        print(f"Network      : {network}")

    if unresolved:
        print(
            "\nWARNING: At least one group could not be "
            "resolved. Verify PAPI hostname fields "
            "and API permissions.",
            file=sys.stderr,
        )
        return 3

    return 0


def main():
    parser = argparse.ArgumentParser(
        description=(
            "Read-only Akamai hostname to "
            "PAPI Access Group lookup"
        )
    )

    parser.add_argument(
        "--section",
        default="DEV",
        help="EdgeGrid .edgerc section",
    )

    parser.add_argument(
        "--edgerc-file",
        default="~/.edgerc",
    )

    parser.add_argument(
        "--fqdn",
        required=True,
    )

    parser.add_argument(
        "--network",
        choices=["PRODUCTION", "STAGING"],
        default="PRODUCTION",
    )

    args = parser.parse_args()

    session, base_url = setup_session(
        args.edgerc_file,
        args.section,
        False,
    )

    if session is None or not base_url:
        raise RuntimeError(
            "EdgeGrid session initialization failed"
        )

    return lookup(
        session,
        base_url,
        args.fqdn,
        args.network,
    )


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as exc:
        print(f"ERROR: {exc}", file=sys.stderr)
        sys.exit(1)
