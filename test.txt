
#!/usr/bin/env python3
"""
Read-only Akamai CPS certificate status diagnostic.

Place alongside akamai_cert_manager_proxy.py.
No POST, PUT, PATCH or DELETE requests are made here.
"""

import argparse
import json
import sys
from datetime import datetime, timezone
from pathlib import Path

from akamai_cert_manager_proxy import setup_session


def fetch(session, base_url, path, accept):
    url = base_url.rstrip("/") + "/" + path.lstrip("/")

    try:
        response = session.get(
            url,
            headers={"Accept": accept},
            timeout=(10, 45)
        )

        result = {
            "http_status": response.status_code,
            "url_path": path
        }

        try:
            result["body"] = response.json()
        except ValueError:
            result["error"] = response.text[:1500]

        if not response.ok:
            result["error"] = (
                result.get("error")
                or "API request failed; inspect response body"
            )

        return result

    except Exception as exc:
        return {
            "url_path": path,
            "error": f"{type(exc).__name__}: {exc}"
        }


def main():
    parser = argparse.ArgumentParser(
        description="Read-only Akamai CPS status check"
    )

    parser.add_argument(
        "--enrollment-id",
        required=True
    )

    parser.add_argument(
        "--section",
        default="DEV"
    )

    parser.add_argument(
        "--edgerc-file",
        default="~/.edgerc"
    )

    parser.add_argument(
        "--output",
        default=None,
        help="Optional JSON output file"
    )

    args = parser.parse_args()

    session, base_url = setup_session(
        args.edgerc_file,
        args.section,
        False
    )

    if session is None or not base_url:
        print(
            "ERROR: EdgeGrid session initialization failed",
            file=sys.stderr
        )
        return 2

    enrollment_id = args.enrollment_id.strip()

    if not enrollment_id.isdigit():
        parser.error("--enrollment-id must be numeric")

    prefix = f"cps/v2/enrollments/{enrollment_id}"

    report = {
        "checked_at_utc": datetime.now(
            timezone.utc
        ).isoformat(),
        "enrollment_id": enrollment_id,
        "section": args.section,
        "read_only": True
    }

    # Enrollment information
    report["enrollment"] = fetch(
        session,
        base_url,
        prefix,
        "application/vnd.akamai.cps.enrollment.v12+json"
    )

    # Staging and production deployments
    report["deployments"] = fetch(
        session,
        base_url,
        prefix + "/deployments",
        "application/vnd.akamai.cps.deployments.v7+json"
    )

    # Renewal/change history
    report["changes"] = fetch(
        session,
        base_url,
        prefix + "/changes",
        "application/vnd.akamai.cps.changes.v2+json"
    )

    changes_body = report["changes"].get("body")

    changes = (
        changes_body.get("changes", [])
        if isinstance(changes_body, dict)
        else []
    )

    if not isinstance(changes, list):
        changes = []

    active_states = {
        "pending",
        "to-be-deployed",
        "deploying",
        "in-progress",
        "waiting-for-approval"
    }

    selected = []

    for index, change in enumerate(changes):
        if not isinstance(change, dict):
            continue

        change_id = (
            change.get("changeId")
            or change.get("id")
        )

        status = str(
            change.get("status", "")
        ).lower()

        if change_id is not None and (
            index == 0 or status in active_states
        ):
            if str(change_id) not in selected:
                selected.append(str(change_id))

    # Fetch detailed status for selected changes
    report["change_details"] = {}

    for change_id in selected[:20]:
        report["change_details"][change_id] = fetch(
            session,
            base_url,
            prefix + "/changes/" + change_id,
            "application/vnd.akamai.cps.change.v2+json"
        )

    print(json.dumps(
        report,
        indent=2,
        default=str
    ))

    if args.output:
        output = Path(args.output)

        output.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        output.write_text(
            json.dumps(
                report,
                indent=2,
                default=str
            ) + "\n",
            encoding="utf-8"
        )

        print(
            f"JSON saved to {output}",
            file=sys.stderr
        )

    if any(
        report[key].get("http_status") != 200
        for key in (
            "enrollment",
            "deployments",
            "changes"
        )
    ):
        print(
            "WARNING: One or more CPS reads failed; "
            "status is incomplete.",
            file=sys.stderr
        )
        return 1

    return 0


if __name__ == "__main__":
    sys.exit(main())
