def check_ad_group_exists(group_name):
    encoded_group = urllib.parse.quote(group_name)

    url = (
        f"{DEFAULT_BASE_URL}?"
        f"serverUrl={DEFAULT_SERVER_URL}"
        f"&baseDN={DEFAULT_BASE_DN}"
        f"&attributes=cn"
        f"&filter=%28cn%3D{encoded_group}%29"
        f"&limit=10&sentFrom=CyberWASP"
    )

    try:
        response = requests.get(url, timeout=10)
        response.raise_for_status()
        data = response.json()

        if data:
            print(f"[FOUND] AD group exists: {group_name}")
            return True

        print(f"[NOT FOUND] AD group: {group_name}")
        return False

    except Exception as e:
        print(f"[ERROR] AD group lookup failed: {e}")
        return None
