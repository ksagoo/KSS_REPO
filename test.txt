import argparse
import sys
from urllib.parse import urljoin

from akamai_cert_manager_proxy import setup_session


def get_json(session, url, **kwargs):
    response = session.get(
        url,
        headers={
            "Accept": "application/json",
            "PAPI-Use-Prefixes": "true",
        },
        timeout=30,
        **kwargs,
    )
    response.raise_for_status()
    return response.json()


def find_group(groups, target_id):
    for group in groups:
        if str(group.get("groupId")) == target_id:
            return group

        found = find_group(
            group.get("subGroups") or [],
            target_id,
        )
        if found:
            return found

    return None


def lookup(session, base_url, fqdn):
    fqdn = fqdn.strip().rstrip(".").lower()

    url = urljoin(
        base_url.rstrip("/") + "/",
        "papi/v1/hostnames",
    )

    matches = []
    params = {
        "hostname": fqdn,
        "limit": 100,
        "network": "PRODUCTION",
    }

    while url:
        data = get_json(session, url, params=params)
        collection = data.get("hostnames") or {}

        for item in collection.get("items", []):
            hostname = (
                item.get("cnameFrom") or ""
            ).rstrip(".").lower()

            if hostname == fqdn:
                matches.append(item)

        next_link = collection.get("nextLink")
        url = (
            urljoin(base_url, next_link)
            if next_link
            else None
        )
        params = None

    if not matches:
        print(f"No active production property found: {fqdn}")
        return 1

    groups = get_json(
        session,
        urljoin(
            base_url.rstrip("/") + "/",
            "identity-management/v3/user-admin/groups",
        ),
    )

    if not isinstance(groups, list):
        raise RuntimeError("Unexpected IAM groups response")

    for item in matches:
        group_id = str(item.get("groupId") or "")
        group = find_group(groups, group_id)

        print(f"\nFQDN         : {fqdn}")
        print(f"Property     : {item.get('propertyName', 'Unknown')}")
        print(f"Property ID  : {item.get('propertyId', 'Unknown')}")
        print(f"Access Group : {group.get('groupName') if group else 'Unresolved'}")
        print(f"Group ID     : {group_id or 'Unknown'}")
        print(f"Contract ID  : {item.get('contractId', 'Unknown')}")
        print("Network      : PRODUCTION")

    return 0


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--section", default="DEV")
    parser.add_argument("--fqdn", required=True)
    parser.add_argument("--edgerc-file", default=None)

    args = parser.parse_args()

    session, base_url = setup_session(
        args.edgerc_file,
        args.section,
        False,
    )

    return lookup(session, base_url, args.fqdn)


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as exc:
        print(f"ERROR: {exc}", file=sys.stderr)
        sys.exit(1)
