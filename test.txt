if args.fqdn:
    log("INFO", f"Resolving Enrollment ID for FQDN: {args.fqdn}")

    try:
        resolved_id = manager.get_enrollment_id_by_fqdn(args.fqdn)

        if not resolved_id:
            raise RuntimeError(
                f"Unable to resolve FQDN: {args.fqdn}"
            )

        if args.enrollment_id and args.enrollment_id != resolved_id:
            raise ValueError(
                "Provided Enrollment ID does not match FQDN"
            )

        args.enrollment_id = resolved_id

    except (ValueError, RuntimeError) as exc:
        log("ERROR", f"FQDN lookup failed: {exc}")
        raise SystemExit(1)
